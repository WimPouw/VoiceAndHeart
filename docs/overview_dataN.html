# ppn 41 and 42 are the SAME participant as ppn 4 (split across IDs).
recode_ppn <- function(df, cols = c("ppn", "pp")) {
  for (col in cols) if (col %in% names(df)) {
    v <- suppressWarnings(as.numeric(df[[col]]))
    v[v %in% c(41, 42)] <- 4
    df[[col]] <- v
  }
  df
}

read_triallist <- function(meta_dir) {
  fs <- list.files(meta_dir, pattern = "triallist*", full.names = TRUE)
  d  <- bind_rows(lapply(fs, function(p) {
    # Read everything as character so per-file type inference can't
    # disagree across files (e.g. `trial` as <chr> in one, <dbl> in another).
    x <- suppressMessages(read_delim(p, delim = ";", show_col_types = FALSE,
                                     col_types = cols(.default = "c")))
    x$ppn <- parse_number(basename(p))
    x
  }))
  d <- recode_ppn(d)
  d$uniquetrial <- paste0(d$trialindex, "_", d$ppn)
  d[!duplicated(d$uniquetrial), ]
}

track <- function(label, df, ppn_col = "ppn") {
  tibble(step           = label,
         N_trials       = if ("uniquetrial" %in% names(df)) length(unique(df$uniquetrial)) else nrow(df),
         N_participants = length(unique(df[[ppn_col]])))
}

voc_tl <- read_triallist(voc_meta)

voc_filter <- voc_tl %>%
  filter(movement_condition == "no_movement",
         vocal_condition    == "vocalize",
         trial_type         == "main")

voc_full <- suppressMessages(read_csv(file.path(voc_procdtot, "fulldata_heartrate.csv"),
                                      show_col_types = FALSE)) %>%
  recode_ppn() %>%
  # `heartrate_EMG_hz` is created in memory in the source `.Rmd` but not
  # written to the CSV. Recreate it here: rectus for ppn 9, pectoralis otherwise.
  mutate(heartrate_EMG_hz  = ifelse(ppn == 9, heartrate_rectusfft_hz,  heartrate_pectoralisfft_hz),
         heartrate_EMG_bpm = ifelse(ppn == 9, heartrate_rectusfft_bpm, heartrate_pectoralisfft_bpm))

voc_hr <- voc_full %>% filter(!is.na(heartrate_EMG_hz))

bind_rows(
  track("1. Full trial list (all conditions)",                 voc_tl),
  track("2. After filter: no-mov, vocalize, main",             voc_filter),
  track("3. Aggregated (`fulldata_heartrate.csv`)",            voc_full),
  track("4. With valid FFT heart rate",                        voc_hr),
  tibble(step = "5. Rows in H1 coherence model (= 2 × step 4)",
         N_trials = 2 * nrow(voc_hr),
         N_participants = length(unique(voc_hr$ppn)))
) %>%
  kable(caption = "Vocalization — trial/participant count at each step") %>%
  kable_styling(full_width = FALSE, bootstrap_options = c("hover", "condensed"))
